\subsection{Group is a groupoid}
\begin{enumerate}[label=(\alph*)]
    \item Make sense of: ``A group is a groupoid with one object''. \begin{answer}\end{answer}
        Note that morphism composition is associative, that \(id_{G}\in\mor(G,G)\), and that \(\mor(G,G)\) is closed.
        Since \(G\), (as a category) is a groupoid, every morphism in \(\mor(G,G)\) is an isomorphism. Therefore for every \(\phi\in\mor(G,G)\), \(\exists!\phi^{-1}\in\mor(G,G)\), so that \[ \phi\circ\phi^{-1}=\phi^{-1}\circ\phi=id_{G} \]
        Then the set \(\mor(G,G)\) with the operation
        \[
            \circ:\mor(G,G)\times\mor(G,G)\to\mor(G,G)
        \]
        admits group structure. (Dropping the condition that \(G\) is a groupoid gives \(\mor(G,G)\) the structure of a monoid.)
    \item Describe a groupoid that is not a group. \begin{answer} \end{answer}
        Let \(\mathcal{C}\) be a groupoid with two objects \(A,B\). Note that in a group, composition of any two elements is defined, however in our category \(\mathcal{C}\), we have things like \(\phi,\psi\in\mor(A,B)\) which you can not compose because the sources and targets do not match.
\end{enumerate} 
\subsection{Automorphism Group} Let \(A\in\obj(\mathcal{C})\). Show that the invertible elements of \(\mor(A,A)\) form a group. (Called the \ul{automorphism group} \(\aut(A)\)). What are the automorphism groups for \(\catname{Sets}\) and \(\catname{Vect}_{k}\)? Prove that isomorphic objects have isomorphic automorphism groups. \begin{answer} \end{answer}
Let \(A\in\obj(\mathcal{C})\) and consider the subset \(\aut(A)\subseteq\mor(A,A)\) with the property \(f\in\aut(A)\iff f\) is invertible. Then notes \(id_{A}\in\aut(A)\), which serves as the identity of the group. As before, morphism composition is associative, and \(\forall f\in\aut(A)\), since \(f\) is invertible, \(\exists f^{-1}\in\aut(A)\) such that \(f\circ f^{-1}=f^{-1}\circ f=id_{A}\). Therefore, \(\aut(A)\) satisfies the group axioms.
\par If \(X\in\obj(\catname{Sets})\), \(\aut(X)\) consists of permutations of the elements of \(X\). These are functions \(f:X\to X\) which are invertible (bijections). If \(V\in\obj(\catname{Vect}_{k})\) is a vector space, \(Aut(V)\) consists of invertible matrices with entries from \(k\).
\begin{proof}
    Let \(A\) and \(B\) be isomorphic objects, so \(\exists f:A\to B\) and \(g:B\to A\) such that \(g\circ f=id_{A}\) and \(f\circ g=id_{B}\). We know \(\aut(A)\) and \(\aut(B)\) are groups. Define the maps
    \[
        \pi:\aut(A)\to\aut(B)
    \]
    by \[
        \pi:\phi\mapsto f\circ\phi
    \]
    and \[
        \pi^{-1}:\aut(B)\to\aut(A)
    \]
    \[
        \pi^{-1}:\psi\mapsto g\circ\psi
    \]
    and then
    \[
        \pi^{-1}(\pi(\phi))=g\circ(f\circ\phi)=(g\circ f)\circ\phi=id_{A}\circ\phi=\phi
    \]

    \[
        \pi(\pi^{-1}(\psi))=f\circ (g\circ \psi)=(f\circ g)\circ\psi=id_{B}\circ\psi=\psi
    \]
    So \(\pi^{-1}\circ\pi=id_{\aut(A)}\) and \(\pi\circ\pi^{-1}=id_{\aut(B)}\). Therefore \(\aut(A)\cong\aut(B)\).
\end{proof}

\subsection{Initials and Terminals}\begin{defintion}
    An object in a category \(\mathcal{C}\) is \ul{initial} if it has exactly 1 map to each object in \(\mathcal{C}\). An object is \ul{terminal} (or \ul{final}) if it has exactly one map \emph{from} each object. An object is a \ul{zero object} if it is both initial and terminal.
\end{defintion}  
Show that any two initial objects are uniquely isomorphic. Same for terminal.
\begin{proof}
    Let \(A,A'\) be two initial objects. Then, since \(A\) is initial, \(\exists!f:A\to A'\). Similarly, since \(A'\) is initial, \(\exists!g:A'\to A\). Then \(f\circ g:A\to A\) is unique, but \(id_{A}:A\to A\) is a map \(A\to A\). So \(f\circ g=id_{A}\). Similarly, \(g\circ f:A'\to A'\) is unique, so \(g\circ f=id_{A'}\). Therefore, \(A\cong A'\) uniquely. For terminal, flip the arrows.
\end{proof}
\subsection{Examples of initals/terminals/zeros} What are initial, terminal, and zero objects in \(\catname{Sets}\), \(\catname{Rings}\), \(\catname{Top}\)?
\begin{answer}
\end{answer}
\ \begin{itemize}
    \item In \(\catname{Sets}\), the empty set \(\emptyset\) maps uniquely into every set \(X\), via the empty function (which acts as the inclusion \(\emptyset\hookrightarrow X\)), so it is inital. However, you can not define a function \(\phi:X\to\emptyset\) for all \(X\), so it is not a terminal object. For any singleton set \(\{a\}\), there is a unique map \(g:X\to\{a\},\ g(x)=a\) for all \(x\in X\). Therefore, each singleton set is terminal.
    \item In \(\catname{Rings}\), the zero ring \(\{0\}\) is terminal, because for any ring \(R\), take \(f:R\to\{0\}\), \(f(r)=0\) for all \(r\in R\). This is a valid ring hom, and it is unique. The ring \((\Z,+,\cdot)\) is initial, because for any ring \(R\), we must have (by our conventions) \[
            f:\Z\to R,\ f:1_{\Z}\mapsto 1_{R}
        \] and since \(\Z\) is generated by \(1_{\Z}\), that rule determines the entire homomorphism. (Note: this map captures the Characterstic of the ring \(R\), because it is the \(n\) in \(\ker{f}=n\Z\)).
    \item In \(\catname{Top}\), similar to \(\catname{Sets}\), the empty space \(\emptyset\) (with its corresponding toplogy) is initial via the empty function, which is trivially continuous. The terminal objects are singletons \(\{a\}\), because there is a unique map \(f:X\to\{a\}\), \(f(x)=a\), which is continuous because the preimage of \(a\) is \(X\), which is open.
\end{itemize}
\subsection{Mapping a ring into its localiztion}Show that \(A\to S^{-1}A\), \((a\mapsto \frac{a}{1})\) is injective iff \(S\) contains no zerodivisors.
\begin{proof}
    \((\implies):\) Let \(f:A\to S^{-1}A\), \(f:a\mapsto \frac{a}{1}\) be injective. Let \(a\neq b\), then \(f(a)\neq f(b)\implies \frac{a}{1}\neq \frac{b}{1}\). That means \[
        \centernot\exists s\in S,\ \text{so that } s(a-b)=0
    \]
    So for \(x=(a-b)\in A\) nonzero, there is no \(s\in S\) so that \(s\cdot x=0\). Therefore, \(S\) contains no zerodivisors.
    \par\null\par \((\impliedby):\) Let \(S\) be a multiplicative set with no zerodivisors. Let \(f:A\to S^{-1}A\), \(f:a\mapsto \frac{a}{1}\) be a function. Let \(f(a)=f(b)\) for some \(a,b\in A\). Then \(\frac{a}{1}=\frac{b}{1}\implies\exists s\in S\) such that \(s(a-b)=0\). Since \(s\) is not a zerodivisor, we must have \(a-b=0\implies a=b\). Therefore \(f\) is injective.
\end{proof}
\subsection{Univseral Property of Localization} Verify \(A\to S^{-1}A\) satisfies the following universal property: \(S^{-1}A\) is initial among \(A\)-algebras \(B\) where every element of \(S\) is sent to an invertible element in \(B\). Translation: any map \(A\to B\) where every \(s\in S\) is sent to an invertible in \(B\) must factor uniquely through \(A\to S^{-1}A\).
\begin{remark}
    An \(A\)-module \(M\) is the same as a map \[
        \rho:A\to\End_{\Z}(M)\]\[\rho: s\mapsto \mu_{s}
        \] where \[
        \mu_{s}:M\to M\] \[
        \mu_{s}:m\mapsto s\cdot m
    \]
    And to say that a scalar \(s\) is invertible is to say that \(\mu_{s}\in\End_{\Z}(M)\) is a bijection (automorphism).
\end{remark}
\begin{proof}
    Let \(f:A\to S^{-1}A\) where \(f(a)=\frac{a}{1}\). Let \(B\) be an \(A\)-algebra, with map \(\phi:A\to B\) such that \(\phi(s)\) is a unit in \(B\). We claim \(\exists!g\) such that 
    \[
        \begin{tikzcd}[row sep=large,column sep=large]
            A \arrow[r, "f"] \arrow[d, "\phi"'] & S^{-1}A \arrow[dl, "!g",dashed] \\
            B &
        \end{tikzcd}
    \]
    commutes. Uniqueness is shown by the following: \[
        g\left(\frac{a}{s}\right)=g(f(a)\cdot f(s)^{-1})=g(f(a))\cdot g(f(s))^{-1}=\phi(a)\cdot\phi(s)^{-1}
    \]
    So what \(g\) does to \(\frac{a}{s}\in S^{-1}A\) is completely determined by \(\phi:A\to B\). To show existence we need to show that \(g\) is well-defined and that it is a ring hom.
    \par
    Well-defined: Let \(\frac{a'}{s'}=\frac{a}{s}\). Then we want \(g\left(\frac{a'}{s'}\right)=g\left(\frac{a}{s}\right)\). \[
        \frac{a'}{s'}=\frac{a}{s}\implies\exists t\in S\ \text{such that } t(a's-s'a)=0
    \]
    This is an expression in \(A\), so apply \(\phi:A\to B\) to both sides:
    \[
        \phi(t)(\phi(a's)-\phi(as'))=0
    \]
    Since \(t\in S\), \(\phi(t)\) is a unit, so it is not a zerodivisor. Therefore \(\phi(a's)=\phi(as')\implies \phi(a')\phi(s)=\phi(a)\phi(s')\). Since \(\phi(s)\) and \(\phi(s')\) are units in \(B\), we can multiply by \(\phi(s)^{-1}\) and \(\phi(s')^{-1}\) to get: \[
        \phi(a')\phi(s')^{-1}=\phi(a)\phi(s)^{-1}\implies g\left(\frac{a'}{s'}\right)=g\left(\frac{a}{s}\right)
    \]
    As required, so \(g\) is well-defined.
    \par Ring homomorphism: We check \(g(1_{S^{-1}A})=1_{B}\): \begin{align*}
        g\left(\frac{1}{1}\right)&=\phi(1_{A})\phi(1_{A})^{-1}\\
                                 &=1_{B}\cdot 1_{B}\\
                                 &=1_{B}\\
    \end{align*}
    \(g\) respects addition:
    \begin{align*}
        g\left(\frac{a}{s}+\frac{b}{t}\right)&=g\left(\frac{at+bs}{st}\right)\\
                                             &=\phi(at+bs)\phi(st)^{-1}\\
                                             &=\left[ \phi(a)\phi(t)+\phi(b)\phi(s) \right]\phi(s)^{-1}\phi(t)^{-1}\\
                                             &=\phi(a)\phi(s)^{-1}+\phi(b)\phi(t)^{-1}\\
                                             &=g\left(\frac{a}{s}\right)+g\left(\frac{b}{t}\right)
    \end{align*}
    \(g\) respects multiplication:
    \begin{align*}
        g\left(\frac{a}{s}\cdot\frac{b}{t}\right)&=g\left(\frac{ab}{st}\right)\\
                                                 &=\phi(ab)\phi(st)^{-1}\\
                                                 &=\phi(a)\phi(b)\phi(s)^{-1}\phi(t)^{-1}\\
                                                 &=\phi(a)\phi(s)^{-1}\phi(b)\phi(t)^{-1}\\
                                                 &=g\left(\frac{a}{s}\right)\cdot g\left(\frac{b}{t}\right)
    \end{align*}
    So \[
        g:S^{-1}A\to B
    \]
    \[
        g:\frac{a}{s}\mapsto\phi(a)\phi(s)^{-1}
    \]
    is a well defined, unique ring homomorphism which makes the diagram commute. Therefore, every map \(\phi:A\to B\) (every \(A\)-algebra \(B\)) factors uniquely through \(S^{-1}A\).
\end{proof}

\subsubsection{Localization of modules, mini problem:}
Let \(M\) be an \(A\)-module. Then we define \(\phi:M\to S^{-1}M\) as being initial among \(A\)-module maps \(M\to N\) such that elements of \(S\) are invertible in \(N\). Or, \(M\to S^{-1}\) is universal among \(A\)-module maps from \(M\) to modules that are actually \(S^{-1}A\)-modules. Mini problem: Define the objects and morphisms in this category to make this precise.
\begin{answer}
\end{answer}
\begin{itemize}
    \item The objects of this category are pairs \((N_{i},\psi_{i})\) where \(N_{i}\) is an \(A\)-module and \(\psi:M\to N_{i}\) is an \(A\)-module homomorphism such that the multiplication map \(s\times -:N_{i}\to N_{i}\) is an automorphism.
    \item The morphisms of this category are: Let \((N_{1},\psi_{1}),\ (N_{2},\psi_{2})\) be two objects of the category. Then they both define maps from M:
        \[
            \begin{tikzcd}[row sep=large,column sep=large]
                M \arrow[r, "\psi_{1}"] \arrow[d, "\psi_{2}"'] & N_{1} \arrow[dl, "\theta",dashed] \\
                N_{2} & 
            \end{tikzcd}
        \]
        The map \(\theta:N_{1}\to N_{2}\) of \(A\)-modules, such that the triangle commutes \((\theta\circ\psi_{1}=\psi_{2})\) is an element of \(\mor((N_{1},\psi_{1}),(N_{2},\psi_{2}))\).
\end{itemize}
\subsection{Localization of Modules} Show that \(\phi:M\to S^{-1}M\) exists by constructing something satisfying the universal property. Hint: define elements of \(S^{-1}M\) to be \(\frac{m}{s}\) where \(m\in M\), \(s\in S\), with equality given by \[
    \frac{m_{1}}{s_{1}}=\frac{m_{2}}{s_{2}}\iff\exists t\in S\mid t(m_{1}s_{2}-m_{2}s_{1})=0
\]
Addition given by \[
    \frac{m_{1}}{s_{1}}+\frac{m_{2}}{s_{2}}=\frac{m_{1}s_{2}+m_{2}s_{1}}{s_{1}s_{2}}
\]
And the \(S^{-1}A\)-module structure (and therefore the \(A\)-module structure) is given by \[
    \frac{a}{s}\cdot \frac{m_{1}}{t}= \frac{a\cdot m_{1}}{st}
\]
\begin{remark}
    The \(A\)-module structure is achieved by ``pulling back along the localization map'' like so: Given \(\psi:S^{-1}A\to\End_{\Z}(S^{-1}M)\), we take \(f:A\to S^{-1}A\) and compose to get \(\rho_{A}:A\to\End_{\Z}(S^{-1}M)\) where \(\rho_{A}=\psi\circ f\)
\end{remark}
\begin{proof}
    Let \(M\) and \(S^{-1}M\) be \(A\)-modules, with \(S^{-1}M\) defined as above. Define the map of \(A\)-modules
    \[
        \phi:M\to S^{-1}M
    \]
    \[
        \phi:m\mapsto \frac{m}{1}
    \]
    We want to show that this map exists and that it satisfies the universal property. The category we are working in is the (full) subcategory of the slice category \(M\), where objects are pairs \((N,\psi)\) where \(\psi:M\to N\) and every element of \(S\) maps to an invertible in \(N\). Morphisms, say between \((N_{1},\psi_{1})\) and \((N_{2},\psi_{2})\) are maps \(\theta:N_{1}\to N_{2}\) which make \[
        \begin{tikzcd}
            M \arrow[r, "\psi_{1}"] \arrow[d, "\psi_{2}"'] & N_{1} \arrow[dl, "\theta",dashed] \\
            N_{2}  &
        \end{tikzcd}
    \]
    commute (so \(\psi_{2}=\theta\circ\psi_{1}\)). So we want to show \(\phi:M\to S^{-1}M\) is initial in this category (or, the pair \((S^{-1}M,\phi)\) is an initial object).
    \par So, let \((N,\psi)\) be an object of this category. We want to show there \(\exists! \theta:S^{-1}M\to N\) such that 
    \[
        \begin{tikzcd}
            M \arrow[r, "\phi"] \arrow[d, "\psi"'] & S^{-1}M \arrow[dl, "\theta",dashed] \\
            N  &
        \end{tikzcd}
    \]
    commutes (\(\psi=\theta\circ\phi\)).
    \par Uniqueness of \(\theta\) is seen by:
    \begin{align*}
        \theta\left(\frac{m}{s}\right)&=\theta\left(\mu_{s}^{-1}\left(\phi(m)\right)\right)\\
                                      &=\mu_{s}^{-1}\left(\theta\left(\phi(m)\right)\right)\\
                                      &=\mu_{s}^{-1}\left(\psi(m)\right)\\
    \end{align*}
    Where the second equality is due to \(\theta\left(\mu_{s}(m)\right)=\mu_{s}\left(\theta(m)\right)\), then multiply by \(\mu_{s}^{-1}\) (which exists) on both sides.
    \par Well-defined: Let \(\frac{m}{s}=\frac{m'}{s'}\), then \(\exists t\in S\) with \(t(ms'-m's)=0\). Apply \(\psi:M\to N\) to both sides:
    \[
        t(\psi(m)s'-\psi(m')s)=0
    \]
    Since \(t\in S\) is a unit, it is not a zerodivisor, so \(s'\cdot\psi(m)=s\cdot\psi(m')\). Since \(s,s'\in S\), \(s\cdot s'\in S\) and so \(\mu_{s\cdot s'}^{-1}\) exists. Multiply by this on both sides:\[
        \mu_{s}^{-1}\psi(m)=\mu_{s'}^{-1}\psi(m')\implies\theta\left(\frac{m}{s}\right)=\theta\left(\frac{m'}{s'}\right)
    \]
    \(\theta\) respects addition: \begin{align*}
        \theta\left(\frac{m_{1}}{s_{1}}+\frac{m_{2}}{s_{2}}\right)&=\theta\left(\frac{m_{1}s_{2}+m_{2}s_{1}}{s_{1}s_{2}}\right)\\
                                                                  &=\mu_{s_{1}s_{2}}^{-1}\left(\psi(m_{1})\mu_{s_{2}}+\psi(m_{2})\mu_{s_{1}}\right)\\
                                                                  &=\mu_{s_{1}}^{-1}\left(\psi(m_{1})\right)+\mu_{s_{2}}^{-1}(\psi(m_{2}))\\
                                                                  &=\theta\left(\frac{m_{1}}{s_{1}}\right)+\theta\left(\frac{m_{2}}{s_{2}}\right)
    \end{align*}
    \(\theta\) respects \(A\)-scalar multiplication:
    \begin{align*}
        \theta\left(a\cdot \frac{m}{s}\right)&=\theta\left(\frac{a\cdot m}{s}\right)\\
                                             &=\mu_{s}^{-1}\left(\psi(a\cdot m)\right)\\
                                             &=\mu_{s}^{-1}\left(a\cdot\psi(m)\right)\\
                                             &=a\cdot \mu_{s}^{-1}\left(\psi(m)\right)\\
                                             &=a\cdot\theta\left(\frac{m}{s}\right)\\
    \end{align*}
    Where the 4th equality is due to \(\mu_{s}^{-1}\) being the inverse of an \(A\)-linear operation, so it is \(A\)-linear.
    \par So \(\theta\) exists and is unique, therefore \((S^{-1}M,\phi)\) is initial in this category.
\end{proof}
\subsection{Some Localization Problems} \begin{enumerate}[label=(\alph*)]
    \item Show that localization commutes with finite products (equivalently, with finite direct sums).
    \item Show that localization commutes with arbitrary direct sums.
    \item Show that localization does not necessarily commute with infinite products: the obvisou map \(S^{-1}(\prod_{i}M_{i})\to\prod_{i}S^{-1}M_{i}\) induced by the universal property is not always an isomorphism. \\(Hint: \((1,\frac{1}{2},\frac{1}{3},\dots)\in\Q\times\Q\times\Q\times\cdots\))
\end{enumerate}
\begin{proof}
    \begin{enumerate}[label=(\alph*).]
        \item We need to show there is a bijective \(A\)-module morphism
            \[
                f:S^{-1}(M_{1}\times\dots\times M_{n})\to S^{-1}M_{1}\times\dots\times S^{-1}M_{n}
            \]
            Since \(S^{-1}(M_{1}\times\dots\times M_{n})\) is a localization, the map
            \[
                \phi:M_{1}\times\dots\times M_{n}\to S^{-1}(M_{1}\times\dots\times M_{n})
            \]
            is initial among \(A\)-module maps out of \(M_{1}\times\dots\times M_{n}\) where \(s\in S\) acts invertibly. Consider the map \[
                \psi:M_{1}\times\dots\times M_{n}\to S^{-1}M_{1}\times\dots\times S^{-1}M_{n}
            \]
            \[
                \psi:(m_{1},\dots,m_{n})\mapsto\left(\psi_{1}(m_{1}),\dots,\psi_{n}(m_{n})\right)
            \]
            Where \(\psi_{i}:M_{i}\to S^{-1}M_{i}\) is the canonical localization map. This is an \(A\)-module homomorphism out of \(M_{1}\times\dots\times M_{n}\). And since \(\mu_{s}\in\End_{\Z}\left(S^{-1}M_{1}\times\dots\times S^{-1}M_{n}\right)\) acts componentwise, and \(s\) acts invertibly on each component \(\phi_{i}(m_{i})\), \(s\) acts invertibly on the product. So \(\mu_{s}^{-1}\in\End_{\Z}\left(S^{-1}M_{1}\times\dots\times S^{-1}M_{n}\right)\). Therefore the pair \((S^{-1}M_{1}\times\dots\times S^{-1}M_{n},\psi)\) is an object of this category. So by the universal property of localization:
            \[
                \begin{tikzcd}
                    M_{1}\times\dots\times M_{n} \arrow[r, "\phi"] \arrow[d, "\psi"'] & S^{-1}(M_{1}\times\dots\times M_{n}) \arrow[dl, "f",dashed] \\
                    S^{-1}M_{1}\times\dots\times S^{-1}M_{n}  &
                \end{tikzcd}
\]
\[
    \exists !f:S^{-1}(M_{1}\times\dots\times M_{n})\to S^{-1}M_{1}\times\dots\times S^{-1}M_{n}
\]
\[
f:\frac{(m_{1},\dots,m_{n})}{s}\mapsto\left(\frac{m_{1}}{s},\dots,\frac{m_{n}}{s}\right)
\]
Which makes the diagram commute. We want to show \(f\) is a bijective \(A\)-module homomorphism.
\par Surjective: Let \(\left(\frac{m_{1}}{s_{1}},\dots,\frac{m_{n}}{s_{n}}\right)\in S^{-1}M_{1}\times\dots\times S^{-1}M_{n}\). Define \(t=\prod_{i}s_{i}\), since \(S\) is multiplicative, \(t\in S\). Let \(x_{i}=\prod_{j\neq i}s_{j}\in S\), then
\[
    \left(\frac{m_{1}}{s_{1}},\dots,\frac{m_{n}}{s_{n}}\right)=\left(\frac{x_{1}m_{1}}{t},\dots,\frac{x_{n}m_{n}}{t}\right)=\theta\left(\frac{(x_{1}m_{1},\dots,x_{n}m_{n})}{t}\right)
\]
So \(\theta\) is surjective.\par Injective: Let \(\theta\left(\frac{(m_{1},\dots,m_{n})}{s}\right)=\theta\left(\frac{(m_{1}',\dots,m_{n}')}{t}\right)\). So \[
    \left(\frac{m_{1}}{s},\dots,\frac{m_{n}}{s}\right)=\left(\frac{m_{1}'}{t},\dots,\frac{m_{n}'}{t}\right)
\]
Then componentwise, \(\frac{m_{i}}{s}=\frac{m_{i}'}{t}\implies \exists u_{i}\in S\) such that
\[
    u_{i}(m_{i}t-m_{i}'s)=0
\]
Then let \(u=\prod_{i}u_{i}\). Clearly \(u\in S\), and \[
    u(m_{i}t-m_{i}'s)=0
\]
For all \(i\in I\) simultaneously. So \[
    u\left[(m_{1},\dots,m_{n})t-(m_{1}',\dots,m_{n}')s\right]=0
\]
Therefore \[
\frac{(m_{1},\dots,m_{n})}{s}=\frac{(m_{1}',\dots,m_{n}')}{t}
\]
As required.
\par Alternatively: Let \(\theta\left(\frac{(m_{1},\dots,m_{n})}{s}\right)=0\). Then \(\frac{m_{i}}{s}=0\) for all \(i\). Therefore \[
\frac{(m_{1},\dots,m_{n})}{s}=0
\]
so \(\ker(\theta)=\{0\}\), and so \(\theta\) is injective.
    \item Let \(\theta:S^{-1}\left(\bigoplus_{i\in I}m_{i}\right)\to\bigoplus_{i\in I}S^{-1}M_{i}\) be the unique \(A\)-module homomorphism given by the universal property of localization, which maps \[
    \theta: \frac{(m_{i})_{i\in I}}{s}\mapsto \left(\frac{m_{i}}{s}\right)_{i\in I}
    \]
    It remains to show that \(\theta\) is bijective.\par Surjective: Let
    \[
        x=(x_{i})_{i\in I}\in\bigoplus_{i\in I}S^{-1}M_{i}
    \]
    be an arbitrary element. Since it is an element of a direct sum, it has finite support. Let \(J\subset I\) be the (finite) subset of \(I\) such that \(x_{j}=\frac{m_{j}}{s_{j}}\neq0\) when \(j\in J\), and \(x_{i}=0\) otherwise. The collection of all such \(x_{j}\) terms defines an isomorphism into \(S^{-1}M_{1}\times\dots\times S^{-1}M_{n}\). Then, we use the argument from \((a)\) to see \(\theta\) is surjective.\par Injective: Let \[
    \theta\left(\frac{(m_{i})_{i\in I}}{s}\right)=0
    \]
   Then \(\frac{m_{i}}{s}=0\) for all \(i\in I\). Let \(J\subset I\) again be the finite subset for which \(m_{j}\neq0\), and \(m_{i}=0\) otherwise. If \(j\in J\), then \[
       m_{j}=0\implies\exists u_{j}\in S\text{ such that }u_{j}m_{j}=0
   \]
   Define \(u=\prod_{j\in J}u_{j}\). Then for all \(i\), \(um_{i}=0\). So we have found a \(u\in S\) such that \(um_{i}=0\) for all \(i\in I\), therefore \(\frac{(m_{i})_{i\in I}}{s}=0\), and so \(\ker\left(\theta\right)=\{0\}\) and \(\theta\) is injective.
\item Let \(A=\Z\), \(S=\Z\setminus\{0\}\). Then \(S^{-1}\Z=\Q\). Consider the element \[
        \left(1,\frac{1}{2},\frac{1}{3},\dots\right)\in\Q\times\Q\times\dots
\]
A valid preimage of this element would have to contain every positive integer (in particular every prime number) as a factor of some \(s\in S\). However that is a contradiction because no integer contains every prime as a factor. So \[
\theta:S^{-1}\left(\prod\Z\right)\to\prod\Q
\]
Is not an isomorphism.
\end{enumerate}
\end{proof}
\subsection{Calculating a Tensor Product} Show that \[
\Z/(10)\otimes_{\Z}\Z/(12)\cong\Z/(2)
\]
\begin{remark}
    Note: Over \(\Z\), the tensor product \(M\otimes_{\Z}N=\{m\otimes n\mid m\in M,n\in N\}\) is completely determined by the pure tensor \(1\otimes 1\), because for any \(m\otimes n\in M\otimes_{\Z}N\), we have \[
        m\otimes n=m(1\otimes n)=mn(1\otimes 1)
    \]
\end{remark}
\begin{proof}
    Let \(x\otimes y\in\Z/(10)\otimes_{\Z}\Z/(12)\). Then, \(x\otimes y=xy(1\otimes 1)\). By Bezout, since \(\gcd( 10,12 )=2\), we can write \(2=10(-1)+12(1)\) So for even integers \(2k\) we have\[
        2k=\left[10(-1)+12(1)\right]k
    \]
    So \begin{align*}
        2k(1\otimes 1)&=\left[10(-1)+12(1)\right]k(1\otimes 1)\\
        &=-10k(1\otimes 1)+12k(1\otimes 1)\\
        &=0+0=0\\
    \end{align*}
    For odd integers \(2k+1\) we have 
    \[
        2k+1=\left[10(-1)+12(1)\right]k+1
    \]
    So \[ 
        2k+1(1\otimes 1)=2k(1\otimes 1)+1(1\otimes 1)=1\otimes 1
    \]
    So there are 2 cases:
   \begin{itemize}
       \item If \(2\mid xy\), then \(xy(1\otimes 1)=0\)
       \item If \(2\centernot\mid xy\), then \(xy(1\otimes 1)=1\otimes 1\)
   \end{itemize} 
   (Furthermore, \((1\otimes 1)+(1\otimes 1)=2(1\otimes1)=0\))\\
   Therefore, \(\Z/(10)\otimes_{Z}\Z/(12)\cong\Z/(2)\).
\end{proof}
\subsection{Right Exactness of Tensor Product}
Show that \((-)\otimes_{A}N\) is a covariant functor \(\catname{Mod_{A}}\to\catname{Mod_{A}}\) and that it is right-exact. So if \[
    \begin{tikzcd} 0 \arrow[r] & M' \arrow[r, "g"] & M \arrow[r, "f"] & M'' \arrow[r] & 0 \end{tikzcd}
\]
Is an exact sequence of \(A\)-modules, then
\[
    \begin{tikzcd} M'\otimes_{A}N \arrow[r, "\overline{g}"] & M\otimes_{A}N \arrow[r, "\overline{f}"] & M''\otimes_{A}N \arrow[r] & 0 \end{tikzcd}
\]
Is also exact.
\begin{proof}
First we will show that \[
    F\coloneqq (-)\otimes_{A}N:\catname{Mod_{A}}\to\catname{Mod_{A}}
\]
is a covariant functor. \(F\) sends an object \(M\in\obj(\catname{Mod_{A}})\) to \(M\otimes_{A}N\in\obj(\catname{Mod_{A}})\). For a morphism of \(A\)-modules \(\phi:M_{1}\to M_{2}\), \(F\) sends \(\phi\) to \(\overline{\phi}:M_{1}\otimes_{A}N\to M_{2}\otimes_{A}N\), where \[
    \overline{\phi}(m_{1}\otimes n)=\phi(m_{1})\otimes n
\]
We check: \begin{itemize}
    \item \(F\) preserves identities:
        \[
            F(id_{M})(m\otimes n)=id_{M}(m)\otimes n=m\otimes n
        \]
        so \(F(id_{M})=id_{F(M))}\).
    \item \(F\) respects composition:
        \[
            F(\phi\circ \psi)(m\otimes n)=(\phi\circ\psi)(m)\otimes n=\phi(\psi(m))\otimes n=F(\phi)(F(\psi)(m\otimes n))
        \]
        So \(F\left(\phi\circ\psi\right)=F(\phi)\circ F(\psi)\).
\end{itemize}
Therefore \(F\) is a (covariant) functor. Now we need to show that \(F\) is right-exact. So let 
\[
\begin{tikzcd} 0 \arrow[r] & M' \arrow[r, "g"] & M \arrow[r, "f"] & M'' \arrow[r] & 0 \end{tikzcd}
\]
Be an exact sequence of \(A\)-modules. That is, \(f\) is a surjection and \(g\) is an injection. We want: \[
    \begin{tikzcd} F(M') \arrow[r, "\overline{g}"] & F(M) \arrow[r, "\overline{f}"] & F(M'') \arrow[r] & 0 \end{tikzcd}
\]
To be exact. That is, \(\overline{g}\) surjects onto \(\ker(\overline{f})\), and \(\overline{f}\) is a surjection.
\begin{enumerate}[label=(\roman*).]
    \item \(\ul{\overline{f}\text{ surjection:}}\) Let \(m''\otimes n\in M''\otimes_{A}N\) be a pure tensor. Then since \(m''\in M''\), and \(f\) surjects onto \(M''\), \(\exists m\in M\) with \(f(m)=m''\). Then \(m''\otimes n=f(m)\otimes n=\overline{f}(m\otimes n)\). So \(\overline{f}\) is a surjection.
    \item \(\ul{\im(\overline{g})\subseteq\ker(\overline{f})\text{:}}\) Let \(x\in \im(\overline{g})\), so \(x=g(m)\otimes n\). But \(\im(g)=\ker(f)\), so \(g(m)\in ker(f)\). Then \[
            \overline{f}(x)=f(g(m))\otimes n=0\otimes n=0
    \]
    So \(x\in\ker(\overline{f})\).
    \item By (i) and first isomorphism theorem, \[
            M\otimes_{A} N/\ker(\overline{f})\cong M''\otimes N
    \]
    By (ii), the inclusion \(\im(\overline{g})\hookrightarrow\ker(\overline{f})\) induces a surjection \[
        M\otimes_{A}N/\im(\overline{g})\twoheadrightarrow M\otimes_{A} N/\ker(\overline{f})
    \]
    Composing gives a map
    \[
        \hat{f}:M\otimes_{A}N/\im(\overline{g})\to M''\otimes_{A} N
    \]
\item \(\ul{h:M''\times N\to M\otimes_{A}N/\im(\overline{g})\text{ well defined:}}\)\\ Define \(h(m'',n)=m\otimes n+\im(\overline{g})\), where \(m\) is the preimage of \(m''\) in \(f\). Let \((m'',n)\in M''\times N\), and choose \(m_{1},m_{2}\) such that \(f(m_{1}=m'')\) and \(f(m_{2})=m''\). Then \[
        \left[m_{1}\otimes n+\im(\overline{g})\right]-\left[m\otimes n+\im(\overline{g})\right]=(m_{1}-m_{2})\otimes n+\im(\overline{g})
\]
But \(f(m_{1}-m_{2})=0\), so \(m_{1}-m_{2}\in\ker(f)\supseteq\im(g)\). So \(m_{1}-m_{2}\in\im(g)\). Therefore \((m_{1}-m_{2})\otimes n+\im(\overline{g})=0+\im(\overline{g})\), so \(h\) is well defined.
\item \(\ul{h\text{ is }A\text{-bilinear}:}\)
    \begin{align*}
        h(m_{1}''+m_{2}'',n) &=(m_{1}+m_{2})\otimes n+\im(\overline{g})\\
                             &=\left[m_{1}\otimes n+\im(\overline{g})\right]+\left[m_{2}\otimes n+\im(\overline{g})\right]\\
  &=h(m_{1}'',n)+h(m_{2}'',n)
    \end{align*}
    So \(h\) is linear in the first term.
    \begin{align*}
        h(m'',n_{1}+n_{2})&=m\otimes(n_{1}+n_{2})+\im(\overline{g})\\
                          &=\left[m\otimes n_{1}+\im(\overline{g})\right]+\left[m\otimes n_{2}+\im(\overline{g})\right]\\
    &=h(m'',n_{1})+h(m'',n_{2})\\
    \end{align*}
    So \(h\) is linear in the second term.
    \[
        h(am'',n)=(am)\otimes n+\im(\overline{g})=a(m\otimes n)+\im(\overline{g})=a\cdot h(m'',n)
    \]
    And similar for \(h(m'',an)\). So \(h\) is \(A\)-bilinear.\\
    Therefore, by the universaal property of tensor products:
    \[
    \begin{tikzcd}[row sep=large,column sep=large]
        M''\times N \arrow[r, "\pi"] \arrow[d, "h"] & M''\otimes_{A}N \arrow[dl, "\exists !\psi",dashed] \\
        M\otimes_{A}N/\im(\overline{g})  &
    \end{tikzcd}
    \]
    Where \(h=\psi\circ\pi\).
\end{enumerate}
Finally, we need to show that \(\psi\) and \(\hat{f}\) are inverses. We already know \(\hat{f}\) is surjective, so it suffices to show \(\psi\circ \hat{f}=id_{M\otimes_{A}N/\im(\overline{g})}\), which implies that \(\hat{f}\) is injective. We check:
\[
    \psi(\hat{f}(m\otimes n+\im(\overline{g})))=\psi(f(m)\otimes n)=m\otimes n +\im(\overline{g})
\]
So \(\hat{f}\) is injective, and therefore bijective. Therefore, \[
    M\otimes_{A}N/\im(\overline{g})\cong M\otimes_{A} N/\ker(\overline{f})
\]
So \(\im(\overline{g})=\ker(\overline{f})\) and the sequence
\[
    \begin{tikzcd} M'\otimes_{A} N \arrow[r, "\overline{g}"] & M\otimes_{A} N \arrow[r, "\overline{f}"] & M''\otimes_{A} N \arrow[r] & 0 \end{tikzcd}
\]
Is exact, so the functor \((-)\otimes_{A}N:Mod_{A}\to Mod_{A}\) is right exact.
\end{proof}
\subsection{Uniqueness Of Universal Pair}
Show that \((t,T)\), where \(t:M\times N\to T\) is bilinear in \(A\), is unique up to unique isomorphism.
\begin{proof}
The category we are working in is the subcategory of the coslice category under \(M\times N\) such that morphisms \(M\times N\to X\) are \(A\)-bilinear. That is, objects are pairs \((\psi,X)\) where \(\psi:M\times N\to X\) is \(A\)-bilinear. Morphisms are the diagonal maps:
\[
\begin{tikzcd}[row sep=large,column sep=large]
M\times N \arrow[r, "\psi_{1}"] \arrow[d, "\psi_{2}"] & X_{1} \arrow[dl, "f"] \\
X_{2} &
\end{tikzcd}
\]
such that \(\psi_{2}=f\circ\psi_{1}\).\\ Let \((t,T)\) and \((t',T')\) be two objects both satisfying the universal property, so \[
\begin{tikzcd}[row sep=large,column sep=large]
M\times N \arrow[r, "t"] \arrow[d, "t'"] & T \arrow[dl, "\exists !f"] \\
T' &
\end{tikzcd}
\qquad
\begin{tikzcd}[row sep=large,column sep=large]
M\times N \arrow[r, "t'"] \arrow[d, "t"] & T' \arrow[dl, "\exists !g"] \\
T &
\end{tikzcd}
\]
Both commute. By the universal property for \((t,T)\), \(\exists !f:T\to T'\) such that \(t'=f\circ t\). Similarly by the universal property for \((t',T')\), \(\exists !g:T'\to T\) such that \(t=g\circ t'\). Then \[
t=g\circ t'=g\circ(f\circ t)=(g\circ f)\circ t
\]
and vice versa. Therefore \(g\circ f\) is a morhism \(T\to T\) satisfying:
\[
\begin{tikzcd}[row sep=large,column sep=large]
M\times N \arrow[r, "t"] \arrow[d,"t"] & T \arrow[dl, "\exists !\psi"] \\
T &
\end{tikzcd}
\]
Where \(\psi\circ t=t\), so \(g\circ f=id_{T}\). Similarly, \(f\circ g=id_{T'}\). Therefore \((t,T)\) is unique up to unique isomorphism.
\end{proof}
\subsection{Univseral Property of Tensor Products}
Show that the construction of 1.2.5 satisfies the universal property of tensor products.
\begin{proof}
Let \(F(M\times N)\) be the free module generated by \(M\times N\). Let \(R\) be the submodule generated by the following relations: \begin{itemize}
    \item \(a(m,n)-(am,n)\)
    \item \(a(m,n)-(m,an)\)
    \item \((m_{1}+m_{2},n)-(m_{1},n)-(m_{2},n)\)
    \item \((m,n_{1}+n_{2})-(m,n_{1})-(m,n_{2})\)
\end{itemize}
Then the quotient module \(F(M\times N)/R\) consists of finite linear combinations of pairs \((m,n)\) subject to those relations.
Let
\[
    \tau:M\times N\to F(M\times N)/R
\]
\[
    \tau(m,n)=\overline{(m,n)}
\]
be the canonical map. We want to show that the pair \(\tau,F(M\times N)/R\) is initial among \(A\)-bilinear maps out of \(M\times N\). In other words, let \(b:M\times N\to B\) be \(A\)-bilinear, then \[
\begin{tikzcd}[row sep=large,column sep=large]
    M\times N \arrow[r, "\tau"] \arrow[d, "b"] & F(M\times N)/R \arrow[dl, "\exists !\overline{b}"] \\
B &
\end{tikzcd}
\]
There is a unique \(\overline{b}:F(M\times N)/R\to B\) such that \(b=\overline{b}\circ\tau\).
\begin{enumerate}[label=(\roman*).]
    \item \ul{Uniqueness:} Let \(f:F(M\times N)/R\to B\) be an \(A\)-linear map with \(b=f\circ\tau\). Then for any pair \((m,n)\) we have \[
            f(\tau((m,n)))=f(\overline{(m,n)})=b(m,n)
    \]
    Any element of \(F(M\times N)/R\) is a finite linear combination of these pairs. Since \(f\) is \(A\)-linear and determined by \(b\), \(f\) is unique.
\item \ul{existence:} We must show \(\overline{b}\) is well defined on \(F(M\times N)/R\), and that \(\overline{b}\) is a valid \(A\)-module homomorphism.
    \begin{enumerate}[label=(\alph*).]
        \item \ul{Well defined:} Let \(\overline{x}=0\) in \(F(M\times N)/R\). By definition, \(x\in R\). Since \(x\in R\), it looks like a finite linear combination of the relations defined above. So then \(\overline{b}(\overline{x})=b(x)\), since \(b\) is an \(A\)-bilinear map, we have that all the relations defining \(R\) hold. Therefore, \(b(x)=0\) and so \(\overline{b}\) is well defined.
        \item \ul{\(\overline{b}\) is an \(A\)-module homomorphism:} Let \(x,y\in F(M\times N)/R\), and let \(a\in A\). 
            \begin{align*}
                \overline{b}\left(\overline{x}+\overline{y}\right)&=\overline{b}\left(\overline{x+y}\right)\\
            &=b\left(x+y\right)\\
            &=b\left(x\right)+b\left(y\right)\\
            &=\overline{b}(\overline{x})+\overline{b}(\overline{y})
            \end{align*}
            and \begin{align*}
                a\overline{b}(\overline{x})&=ab(x)\\
            &=b(ax)\\
            &=\overline{b}(\overline{ax})\\
            &=\overline{b}(a\overline{x})\\
            \end{align*}
    \end{enumerate}
    So \(\overline{b}\) is a valid \(A\)-module homomorphism.
\end{enumerate}
Therefore, \((\tau,F(M\times N)/R)\) is initial among \(A\)-bilinear maps out of \(M\times N\).
\end{proof}
\subsection{Extension of Scalars}
\begin{enumerate}[label=(\alph*).]
    \item Let \(A\) be a ring, let \(M\) be an \(A\)-module, and let \(f:A\to B\) be a ring homomorphism. We know \(B\otimes_{A}M\) is an \(A\)-module. Show that it is also a \(B\)-module, and that this defines a functor from \(Mod_{A}\to Mod_{B}\).
\begin{proof}
We need a function \(\phi:B\times(B\otimes_{A}M)\to B\otimes_{A}M\) which satisfies the \(B\)-module axioms. We will do so by constructing maps \(B\otimes_{A}M\to B\otimes_{A}M\) for each \(b\in B\). Fix \(b\in B\) and consider the map\pagebreak \[
\psi_{b}:B\times M\to B\otimes_{A}M
\]
\[
\psi_{b}(b',m)=bb'\otimes m
\]
\begin{enumerate}[label=(\roman*).]
    \item Linearity in the first term: 
        \begin{align*}
        \psi_{b}(b'_{1}+b'_{2},m)&=b(b'_{1}+b'_{2})\otimes m\\
        &=bb'_{1}\otimes m+bb'_{2}\otimes m\\
        &=\psi_{b}(b'_{1},m)+\psi_{b}(b'_{2},m)\\
        \end{align*}
    \item Linearity in the second term:
        \begin{align*}
        \psi_{b}(b',m_{1}+m_{2})&=bb'\otimes (m_{1}+m_{2})\\
        &=bb'\otimes m_{1}+bb'\otimes m_{2}\\
        &=\psi_{b}(b',m_{1})+\psi_{b}(b',m_{2})\\
        \end{align*}
    \item Scalar multiplication:
        \begin{align*}
        \psi_{b}(ab',m)&=bab'\otimes m\\
                       &=abb'\otimes m\tag{Rings commutative by convention}\\
                       &=bb'\otimes am\\
                       &=a(bb'\otimes m)\\
        &=a\cdot \psi_{b}(b',m)\\
        \end{align*}
\end{enumerate}
So \(\psi_{b}\) is \(A\)-bilinear for all \(b\in B\). By the universal property of tensor products, each \(\psi_{b}\) factors uniquely through \(B\otimes_{A}M\). So for each \(b\in B\), there is an induced map:
\[
\phi_{b}:B\otimes_{A}M\to B\otimes_{A}M
\]
\[
\phi_{b}(b'\otimes m)=bb'\otimes m
\]
So using these \(\phi_{b}\), we can define \[
\phi:B\times(B\otimes_{A}M)\to B\otimes_{A}M
\]
\[
\phi(b,b'\otimes m)=\phi_{b}(b'\otimes m)
\]
Then we check:
\begin{enumerate}[label=(\roman*).]
    \item Identity: \begin{align*}
            \phi(1_{B},b'\otimes m)&=\phi_{1_{B}}(b'\otimes m)\\
                                   &=(1_{B}b')\otimes m\\
                                   &=b'\otimes m\\
    \end{align*}
    \item Linearity in first term:
        \begin{align*}
        \phi(b_{1}+b_{2},b'\otimes m)&=\phi_{b_{1}+b_{2}}(b'\otimes m)\\
                                     &=\left[(b_{1}+b_{2})b'\right]\otimes m\\
                                     &=b_{1}b'\otimes m+b_{2}b'\otimes m\\
                                     &=\phi(b_{1},b'\otimes m)+\phi(b_{2},b'\otimes m)\\
        \end{align*}
    \item Linearity in second term:
        \begin{align*}
            \phi(b,b'_{1}\otimes m_{1}+b'_{2}\otimes m_{2})&=\phi_{b}(b'_{1}\otimes m+b'_{2}\otimes m)\\
        &=\phi_{b}(b'_{1}\otimes m)+\phi_{b}(b'_{2}\otimes m)\\
        &=\phi(b,b'_{1}\otimes m)+\phi(b,b'_{2}\otimes m)\\
        \end{align*}
    \item \(B\)-scalar multiplication:
        \begin{align*}
        \phi(b_{1}b_{2},b'\otimes m)&=\phi_{b_{1}b_{2}}(b'\otimes m)\\
        &=(b_{1}b_{2})b'\otimes m\\
        &=b_{1}(b_{2}b'\otimes m)\\
        &=b_{1}(\phi_{b_{2}}(b'\otimes m))\\
        &=\phi_{b_{1}}(\phi_{b_{2}}(b'\otimes m))\\
        &=\phi(b_{1},\phi(b_{2},b'\otimes m))\\
        \end{align*}
\end{enumerate}
So \(\phi\) is a valid \(B\)-module map, so \(B\otimes_{A}M\) is a \(B\)-module. Therefore, for any \(A\)-module \(M\) (any object of \(\catname{Mod_{A}}\)) we can construct a functor \[
    F:\catname{Mod_{A}}\to\catname{Mod_{B}}
\]
So that \(F(M)\coloneqq B\otimes_{A}M\) is the object assignment, and for an \(A\)-module morphism \(g:M\to N\), we assign a map \(F(g)=\overline{g}:B\otimes_{A}M\to B\otimes_{A}N\). We need to show that \(\overline{g}\) exists and is unique.\\
So let \(g:M\to N\) be an \(A\)-module morphism. Define the map: \[
    \hat{g}:B\times M\to B\otimes_{A}N
\]
\[
    \hat{g}(b,m)=b\otimes g(m)
\]
We claim \(\hat{g}\) is \(A\)-bilinear. We check:
\begin{enumerate}[label=(\roman*).]
    \item Linearity in the first term
       \begin{align*}
           \hat{g}(b_{1}+b_{2},m)&=b_{1}+b_{2}\otimes g(m)\\
       &=b_{1}\otimes g(m)+b_{2}\otimes g(m)\\
       &=\hat{g}(b_{1},m)+\hat{g}(b_{2},m)\\
       \end{align*} 
    \item Linearity in the second term
        \begin{align*}
            \hat{g}(b,m_{1}+m_{2})&=b\otimes g(m_{1}+m_{2})\\
                                  &=b\otimes g(m_{1})+g(m_{2})\tag{By linearity of \(g\)}\\
        &=b\otimes g(m_{1})+b\otimes g(m_{2})\\
        &=\hat{g}(b,m_{1})+\hat{g}(b,m_{2})\\
        \end{align*}
    \item \(A\)-scalar multiplication:
        \begin{align*}
            a\hat{g}(b,m)&=a\cdot(b\otimes g(m))\\
        &=ab\otimes g(m)\\
        &=\hat{g}(ab,m)\\
        &=b\otimes ag(m)\\
        &=b\otimes g(am)\\
        &=\hat{g}(b,am)
        \end{align*}
\end{enumerate}
Therefore \(\hat{g}\) is an \(A\)-bilinear map out of \(B\times M\). By the universal property of tensor products, there is a unique \(A\)-module morphism \(\overline{g}:B\otimes_{A}M\to B\otimes_{A}N\). We need to show that \(\overline{g}\) is actually a \(B\)-module morphism. We check:
\begin{align*}
    \overline{g}(b\cdot b'\otimes m)&=bb'\otimes g(m)\\
&=b(b'\otimes g(m))\\
&=b\overline{g}(b'\otimes m)\\
\end{align*}
and since \(\overline{g}\) is an \(A\)-module morphism, we have \(\overline{g}(x+y)=\overline{g}(x)+\overline{g}(y)\). So \(\overline{g}\) is a \(B\)-module morphism. So the functor \(F\) maps:\[
    F:\catname{Mod_{A}}\to\catname{Mod_{B}}
\]
\[
    F(M)=B\otimes_{A} M,\qquad F(g)=\overline{g}
\]
Finally, we need to check that \(F\) is a functor:
\begin{enumerate}[label=(\roman*).]
    \item Identity: \(F(id_{M})=\overline{id_{M}}\), which is \[
            \overline{id_{M}}(b\otimes m)=b\otimes id_{M}(m)=b\otimes m
    \]
    So \(F(id_{M})=id_{B\otimes_{A} M}=id_{F(M)}\)
\item Composition: \(F(f\circ g)=\overline{f\circ g}\) and \(F(f)\circ F(g)=\overline{f}\circ\overline{g}\).
    \[
        (\overline{f\circ g})(b\otimes m)=b\otimes(f\circ g)(m)=b\otimes f(g(m))
    \]
    \[
        (\overline{f}\circ\overline{g})(b\otimes m)=\overline{f}(b\otimes g(m))=b\otimes f(g(m))
    \]
   So \(F(f\circ g)=F(f)\circ F(g)\)
\end{enumerate}
Therefore \(F\) is a functor.
\end{proof}
\item If \(A\to C\) is another morphism of rings, show that \(B\otimes_{A}C\) has a natural structure of a ring. Hint: Multiplication will be given by \[
        (b_{1}\otimes c_{1})(b_{2}\otimes c_{2})=(b_{1}b_{2})\otimes(c_{1}c_{2})
\]
\begin{proof}
Since \(h:A\to C\) is a ring homomorphism, \(C\) has the structure of an \(A\)-module by \(a\cdot b=h(a)b\). Therefore by (a), we know \(B\otimes_{A}C\) has the structure of an \(A\)-module. Therefore \(B\otimes_{A}C\) is an abelian group. It remains to show that there is a multiplication on \(B\otimes_{A}C\) which satisfies the ring axioms:
\begin{enumerate}[label=(\roman*).]
    \item \(\exists 1_{B\otimes_{A}C}\in B\otimes_{A}C\) with \(1_{B\otimes_{A}C}\cdot x=x\cdot 1_{B\otimes_{A}C}=x\)
    \item \(x\cdot(y+z)=x\cdot y+x\cdot z\) for \(x,y,z\in B\otimes_{A}C\)
    \item \(x\cdot(y\cdot z)=(x\cdot y)\cdot z\) for \(x,y,z\in B\otimes_{A}C\)
\end{enumerate}
Let 
\[
\cdot:B\otimes_{A}C\times B\otimes_{A}C\to B\otimes_{A}C
\]
\[
    \cdot:\left[(b_{1}\otimes c_{1}),(b_{2}\otimes c_{2})\right]\mapsto(b_{1}b_{2})\otimes(c_{1}c_{2})
\]
Using the multiplication in \(B\) and \(C\), respectively.\\
\textbf{Proof of (i):} Since \(B\) and \(C\) are rings, by our convention \(\exists 1_{B}\in B\) and \(1_{C}\in C\). Let \(1_{B}\otimes 1_{C}\in B\otimes_{A}C\). We check: \[
    (1_{B}\otimes 1_{C})\cdot (b\otimes c)=(1_{B}b)\otimes(1_{C}c)=b\otimes c
\]
So \(1_{B}\otimes 1_{C}\) is the identity in \(B\otimes_{A}C\).\\
\textbf{Proof of (ii):} Let \(b_{1},b_{2},b_{3}\in B\) and \(c_{1},c_{2},c_{3}\in C\).
\begin{align*}
    (b_{1}\otimes c_{1})\cdot\left[b_{2}\otimes c_{2}+b_{3}\otimes c_{3}\right]&=(b_{1}\otimes c_{1})\cdot(b_{2}+b_{3}\otimes c_{2}+c_{3})\\
&=(b_{1}(b_{2}+b_{3}))\otimes c_{1}(c_{2}+c_{3})\\
&=b_{1}b_{2}+b_{1}b_{3}\otimes c_{1}c_{2}+c_{1}c_{3}\\
\end{align*}
And
\begin{align*}
    b_{1}\otimes c_{1}\cdot b_{2}\otimes c_{2}+b_{1}\otimes c_{1}\cdot b_{3}\otimes c_{3}&=b_{1}b_{2}\otimes c_{1}c_{2}+b_{1}b_{3}\otimes c_{1}c_{3}\\
&=b_{1}b_{2}+b_{1}b_{3}\otimes c_{1}c_{2}+c_{1}c_{3}\\
&=(b_{1}\otimes c_{1})\cdot\left[b_{2}\otimes c_{2}+b_{3}\otimes c_{3}\right]
\end{align*}
So multiplication distributes over \(+\).\\
\textbf{Proof of (iii):} Let \(b_{1},b_{2},b_{3}\in B\) and \(c_{1},c_{2},c_{3}\in C\).
\begin{align*}
    (b_{1}\otimes c_{1})\cdot[(b_{2}\otimes c_{2})\cdot(b_{3}\otimes c_{3})]&=(b_{1}\times c_{1})\cdot(b_{2}b_{3}\otimes c_{2}c_{3})\\
&=b_{1}(b_{2}b_{3})\otimes c_{1}(c_{2}c_{3})\\
&=(b_{1}b_{2})b_{3}\otimes (c_{1}c_{2})c_{3}\\
&=[(b_{1}\otimes c_{1})\cdot(b_{2}\otimes c_{2})]\cdot(b_{3}\otimes c_{3})\\
\end{align*}
So \(\cdot\) is associative. Therefore, \(B\otimes_{A}C\) has the strcture of a ring.
\end{proof}
\end{enumerate}

\subsection{Tensoring Localization with Module is Localized Module}
If \(S\) is a multiplicative subset of \(A\) and \(M\) is an \(A\)-module, describe a natural isomorphism \[
    (S^{-1}A)\otimes_{A}M\overset{\sim}{\to}S^{-1}M
\]
As \(A\)-modules and as \(S^{-1}A\)-modules.
\begin{proof}
Let \[
    f:(S^{-1}A)\times M\to S^{-1}M
\]
\[
f:\left(\frac{a}{s},m\right)\mapsto \frac{a\cdot m}{s}
\]
be a map. We need to show \(f\) is \(A\)-bilinear, so we check
\begin{enumerate}[label=(\roman*).]
    \item Linearity in the first term
       \begin{align*}
           f\left(\frac{a}{s},m\right)+f\left(\frac{a'}{s'},m\right)&=\frac{ma}{s}+\frac{ma'}{s'}\\
                                                                    &=\frac{s'(m'a)+s(ma')}{ss'}\\
       &=\frac{(sa'+s'a)m}{ss'}\\
       &=f\left(\frac{a}{s}+\frac{a'}{s'},m\right)\\
       \end{align*}
    \item Linearity in the second term
        \begin{align*}
        f\left(\frac{a}{s},m+m'\right)&=\frac{a(m+m')}{s}\\
        &=\frac{am+am'}{s}\\
        &=\frac{am}{s}+\frac{am'}{s}\\
        &=f\left(\frac{a}{s},m\right)+f\left(\frac{a}{s},m'\right)\\
        \end{align*}
    \item \(A\)-scalar multiplication
        \begin{align*}
        f\left(c\cdot \frac{a}{s},m\right)&=\frac{c\cdot am}{s}\\
        &=c\cdot\frac{am}{s}\\
        &=c\cdot f\left(\frac{a}{s},m\right)\\
        \end{align*}
        And
        \begin{align*}
        f\left(\frac{a}{s},c\cdot m\right)&=\frac{c\cdot am}{s}\\
        &=c\cdot f\left(\frac{a}{s},m\right)\\
        \end{align*}
\end{enumerate}
So \(f\) is \(A\)-bilinear, and therefore by the universal property of tensor products:
\[
\begin{tikzcd}[row sep=large,column sep=large]
    (S^{-1}A)\times M \arrow[r, "\pi"] \arrow[d, "f"] & (S^{-1}A)\otimes M \arrow[dl, "\exists !\hat{f}",dashed] \\
S^{-1}M &
\end{tikzcd}
\]
There exists a unique \(\hat{f}:(S^{-1}A)\otimes M\to S^{-1}M\) such that \(f=\hat{f}\circ\pi\). Therefore \[
    \hat{f}\left(\frac{a}{s}\otimes m\right)=\frac{am}{s}
\]
It remains to show that \(\hat{f}\) is an isomorphism. Define \[
g:S^{-1}M\to (S^{-1}A)\otimes M
\]
\[
g:\frac{m}{s}\mapsto \frac{1}{s}\otimes m
\]
We need to show that \(g\) is well defined, that it is a valid module homomorphism, and that it is the inverse of \(\hat{f}\).
\begin{itemize}
    \item \ul{Well defined:} Let \(\frac{m}{s}=\frac{m'}{s'}\in S^{-1}M\). Then by definition of equality of fractions, \(\exists t\in S\) so that \(t(s'm-sm')=0\) in \(M\).
        \begin{align*}
        g\left(\frac{m}{s}\right)-g\left(\frac{m'}{s'}\right)&=\left(\frac{1}{s}\otimes m\right)-\left(\frac{1}{s'}\otimes m'\right)\\
        &=\frac{1}{ss'}\otimes s'm-\frac{1}{s's}\otimes sm'\\
        &=\frac{1}{ss'}\otimes (s'm-sm')\\
        &=\frac{1}{tss'}\otimes t(s'm-sm')
        &=\frac{1}{tss'}\otimes 0=0
        \end{align*}
        So \(g\left(\frac{m}{s}\right)=g\left(\frac{m'}{s'}\right)\).
    \item \ul{Module homomorphism:} Need to show \(g\) preserves addition and \(A\)-scalar multiplication.
        \begin{align*}
        g\left(\frac{m}{s}\right)+g\left(\frac{m'}{s'}\right)&=\left(\frac{1}{s}\otimes m\right)+\left(\frac{1}{s'}\otimes m'\right)\\
        &=\frac{s'}{ss'}\otimes m+\frac{s}{s s'}\otimes m'\\
        &=\frac{1}{s s'}\otimes s'm+\frac{1}{s s'}\otimes sm'\\
        &=\frac{1}{s s'}\otimes s'm+sm'=g\left(\frac{m}{s}+\frac{m'}{s'}\right)
        \end{align*}
        and 
        \begin{align*}
        g\left(a\cdot \frac{m}{s}\right)&=g\left(\frac{am}{s}\right)\\
        &=\frac{1}{s}\otimes am\\
        &=a(\frac{1}{s}\otimes m)=a\cdot g\left(\frac{m}{s}\right)\\
        \end{align*}
        So \(g\) is an \(A\)-module homomorphism.
\end{itemize}
Finally, we need to show that \(g\) and \(\hat{f}\) are inverses, so \[
    g\left(\hat{f}\left(\frac{a}{s}\otimes m\right)\right)=g\left(\frac{am}{s}\right)=\frac{1}{s}\otimes am=\frac{a}{s}\otimes m
\]
and 
\[
    \hat{f}\left(g\left(\frac{m}{s}\right)\right)=\hat{f}\left(\frac{1}{s}\otimes m\right)=\frac{m}{s}
\]
Therefore \(\hat{f}\) and \(g\) are inverses, so \(\hat{f}\) is an isomorphism
\[
    (S^{-1}A)\otimes_{A}M\to S^{-1}M
\]
\end{proof}
